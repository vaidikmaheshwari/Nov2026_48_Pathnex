ansible deploy lambda functions using AWS CLI
- name: deploy AWS Lambda Function
  hosts: localhost
  tasks:
    - name: create Lambda function
      command: aws lambda create-function --function-name vaidik-lambda --runtime python3.8 --role arn:aws:iam::124654732:role/lambda-execution-role --handler lambda_function.lambda_handler --zip-file fileb://lambda.zip


terraform - serverless api gateway with lambda integration_http_method
resource "aws_api_gateway_rest_api" "vaidik_api" {
   name        = "vaidik-api"
   description = "Vaidik API"
}
resource "aws_lambda_function" "vaidik_lambda" {
  function_name = "vaidik-lambda"
  runtime       = "python3.8"
  role          = aws_iam_role.lambda_exec_role.arn
  handler       = "lambda_function.lambda_handler"
  filename      = "lambda.zip"
}
resource "aws_api_gateway_integration" "lambda_integration" {
  rest_api_id = aws_api_gateway_rest_api.vaidik_api.id
  resource_id = aws_api_gateway_resource.vaidik_resource.id
  http_method = aws_api_gateway_method.vaidik_method.http_method
  integration_http_method = "POST"
  type = "AWS_PROXY"
  uri = aws_lambda_function.vaidik_lambda.invoke_arn
}

jenkinsfile = deploy serverless applications with lambda
pipeline {
    agent any 
    stages {
       stage('Build') {
          steps {
              echo 'building lambda function...'
              sh 'zip -r lambda.zip lambda/'
          }
       }
       stage('deploy to aws lambda'){
          steps {
              script {
                  sh 'aws lambda update-function-code --function-name vaidik-lambda --zip-file fileb://lambda.zip'
              }
          }
       }
    }
}

Docker
#multi-stage docker build
#stage 1 - build stage
FROM maven:3.9-eclipse-temurin-17 AS builder
WORKDIR /opt/vaidik/java-build
COPY . /opt/vaidik-java-build/
RUN mvn clean package 


#stage 2 production stage
FROM tomcat:9
WORKDIR /usr/local/tomcat/webapps
COPY --from=builder /opt/vaidik/java-build/target/*.war /usr/local/tomcat/webapps/
EXPOSE 8080
CMD ["catalina.sh","run"]
