ansible - setup docker container for nginx
- name: setup docker container for nginx
  hosts: all
  become: yes
  tasks:
    - name: Pull nginx image
      docker_image:
         name: nginx
         source: pull
    - name: run nginx container
      docker_container:
         name: nginx-container
         image: nginx
         state: started
         published_ports:
            - "8080:80"
          


terraform ec2 with auto scaling Group
resource "aws_launch_configuration" "vaidik" {
  name          = "vaidik-config"
  image_id      = "random"
  instance_type = "t3.medium"
}
resource "aws_autoscaling_group" "vaidik" {
  desired_capacity     = 3
  max_size             = 5
  min_size             = 2
  vpc_zone_identifier  = ["subnet-12345678"]
  launch_configuration = aws_launch_configuration.vaidik.id
}


kubernetes - horizontal pod autoscaler (hpa)
apiVersion: apps/v1
kind: Deployment
metadata: 
  name: vaidik-Deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: vaidik-app
  template:
    metadata:
      labels:
         app: vaidik-app
    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
             - containerPort: 80

---


apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: vaidik-hpa
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: deployment
    name: vaidik-deployment
  minReplicas: 1
  maxReplicas: 10
  targetCPUUtilizationPercentage: 50

   

jenkinsfile - deploy to kubernetets
pipeline {
    agent any 
    stages {
       stage('deploy to k8s') {
          steps {
              script {
                  sh 'kubectl apply -f deployment.yaml'
              }
          }
       }
    }
}
