

day -7 automating cloud infrastructure with terraform and ansible
ansible - setup nginx with custom config
- name: setup nginx with custom config
  hosts: all
  become: yes
  tasks:
    - name: install nginx
      yum: 
        name: nginx
        state: present
    - name: copy custom nginx.conf
      copy:
        src: nginx.conf
        dest: /etc/nginx/nginx.conf
    - name: start nginx service
      service:
        name: nginx
        state: started
        enabled: yes




terraform - provision ec2 with iam role
resource "aws_iam_role" "ec2_role" {
  name               = "vaidik_ec2_role"
  assume_role_policy = jsonencode({
    Version = "2012-10-17"
    Statement = [
      {
        Action = "sts:AssumeRole"
        Effect = "Allow"
        Principal = {
          Service = "ec2.amazonaws.com"
        }
      }
    ]
  })
}


resource "aws_instance" "vaidik-ec2" {
  ami                  = "random"
  instance_type        = "t2.medium"
  iam_instance_profile = aws_iam_role_ec2_role.name
  tags = {
    Name = "Vaidik-Ec2"
  }
}
