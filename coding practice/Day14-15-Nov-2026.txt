ansible setup nginx with reverse proxy
```yaml
- name: setup nginx with reverse proxy
  hosts: all
  become: yes 
  tasks:
    - name: install nginx
      yum:
        name: nginx
        state: present
    - name: configure nginx as reverse proxy
      template:
        src: reverse-proxy.conf.j2
        dest: /etc/nginx/conf.d/reverse-proxy.conf
    - name: restart nginx service
      service:
        name: nginx
        state: restarted
        enabled: yes
  


terraform create s3 bucket with versioning and logging

   bucket = "vaidik-bucket"

     enabled = true

kubernetes - ingress with nginx Controller
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: vaidik-ingress
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /
spec:
  rules:
    - host: vaidik.local
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: vaidik-service
                port:
                  number: 80

jenkinsfile - build test and deploy with stages
pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building docker image...'
                docker.build('vaidik-web0-app')
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }
        stage('Deploy to k8s'){
            steps {
               script {
                 sh 'kubectl apply -f kubernetes/deployment.yaml'
               }
            }
        }
    }
}



github ci/cd multistage deployment with helm
stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - dokcer build -t vaidik-web-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "CI_REGISTRY_PASSWORD"
    - dcoker push vaidik-web-app

deploy:
  stage: deploy
  script:
    - helm upgrade --install vaidik-nginx vaidik/nginx-ingress
