day 2 service & tags
ansible - install and enable nginx
- name: Install and start Nginx on Pathnex
  hosts: all
  become: yes

  tasks:
    - name: install nginx
      yum:
        name: nginx
        state: present

    - name: enable nginx
      service: 
        name: nginx
        state: started
        enabled: yes

Kubernetes deployement with 2 Replicas
apiVersion: apps/v1
kind: Deployment
metadata:
  name: vaidik-Deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: vaidik-app
  template:
    metadata:
      labels:
        app: vaidik-app
    spec: 
      containers:
        - name: app
          image: nginx
          ports:
            - containerPorts: 80
