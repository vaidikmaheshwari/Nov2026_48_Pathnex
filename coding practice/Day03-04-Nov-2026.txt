kubernetes - create a replicaset with 3 replicas
apiVersion: apps/v1
kind: ReplicaSet
metadata:
  name: vaidik-replicaset
spec:
  replicas: 3
  selector: 
    matchLabels:
      app: vaidik-app
  template:
    metadata:
      labels:
        app: vaidik-app
    spec:
      containers:
        - name: app
          image: nginx
          ports:
            - containerPort: 80
